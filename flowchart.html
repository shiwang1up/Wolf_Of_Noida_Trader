Wolf of Noida Trader - API Flowchart

graph TD
    %% EXTERNAL DATA SOURCES
    subgraph External_APIs [External Data Sources]
        CD_M[CoinDCX Markets]
        CD_C[CoinDCX Candles]
        CD_O[CoinDCX Orderbook]
        FGI[Fear & Greed Index]
        CP[CryptoPanic News]
        CC[CryptoCompare Social/News]
    end

    %% DATABASE
    subgraph Database [PostgreSQL / Prisma]
        DB_M[(Market Table)]
        DB_C[(Candle Table)]
        DB_S[(Sentiment Table)]
        DB_N[(News Table)]
        DB_O[(Orderbook Snapshot)]
        DB_LZ[(Liquidity Zones)]
        DB_SIG[(Signals Table)]
    end

    %% BACKGROUND PROCESSES
    subgraph Scheduler [Data Scheduler]
        S_M[Market Sync] -->|Upsert| DB_M
        S_C[Candle Polling] -->|Upsert| DB_C
        S_S[Sentiment Polling] -->|Create| DB_S
        S_N[News Polling] -->|Create| DB_N
        S_O[Orderbook Snapshot] -->|Create| DB_O
        S_LZ[Liquidity Analysis] -->|Analyze| DB_LZ
    end

    %% API FLOW
    subgraph API_Endpoints [REST API]
        R_L_SIG[GET /signals/latest]
        R_G_SIG[POST /signals/generate]
        R_A_MKT[GET /analytics/market]
        R_A_LIQ[GET /analytics/liquidity]
    end

    %% AI ENGINE
    subgraph AI_Engine [AI Reasoning Engine]
        A_AGG[Data Aggregator]
        A_IND[Indicator Processor]
        A_LLM[LLM Reasoning - Groq/Gemini/OpenAI]
        A_SAVE[Signal Persistence]
    end

    %% CONNECTIONS: EXTERNAL TO SCHEDULER
    CD_M --> S_M
    CD_C --> S_C
    CD_O --> S_O
    FGI --> S_S
    CP --> S_N
    CC --> S_N

    %% CONNECTIONS: AI ENGINE FLOW
    R_G_SIG --> A_AGG
    S_C -->|Trigger| A_AGG
    A_AGG -->|Fetch| DB_C
    A_AGG -->|Fetch| DB_S
    A_AGG -->|Fetch| DB_N
    A_AGG -->|Fetch| DB_LZ
    A_AGG -->|Live Fetch| CD_O
    
    DB_C --> A_IND -->|Technical Indicators| A_LLM
    A_AGG -->|Context Payload| A_LLM
    A_LLM -->|Signal Proposal| A_SAVE
    A_SAVE -->|Create| DB_SIG

    %% CONNECTIONS: FRONTEND API READS
    DB_SIG --> R_L_SIG
    DB_C --> R_A_MKT
    DB_LZ --> R_A_LIQ

    class R_L_SIG,R_G_SIG,R_A_MKT,R_A_LIQ api;
    class A_AGG,A_IND,A_LLM,A_SAVE ai;
